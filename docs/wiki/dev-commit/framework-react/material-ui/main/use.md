# Использование

#### Варианты импортов

```js
import { Button } from "@mui/material"
import Button from "@mui/material/Button"
```

#### TypeScript

```js
import { ButtonProps } from "@mui/material"
```

## Простой пример

```js
import { Button } from "@mui/material"

const App = () => {
  return (
    <Button variant="contained" color="primary">
      Click
    </Button>
  );
}

export default App
```
