# Команды

::: info
https://help.ubuntu.ru/wiki/командная_строка
:::

```bash
rmdir моя_папка # удаление пустой директории
rm имя_файла    # удаление файла
rm -R моя_папка # удаление директории с файлами (-r - опция рекурсивного удаления )
```

```bash
wget name_url                     # скачивание
sudo apt-get install name_package # скачивание с репозитория и установка
sudo apt-get remove name_package  # удаление пакета
sudo dpkg -i name_package         # установка скачанного пакета с расширением .deb
sudo apt-get update               # обновление
```

```bash
win+пробел # смена языка
```
